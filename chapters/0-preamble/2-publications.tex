\begin{publications}
\begin{itemize}
    \item \textbf{Section~\ref{sec:concise_L0}: A concise proof of the $\mathbb{L}_0$ dichotomy}\\
    Contains original proofs first disseminated at the Fields Institute Set Theory Seminar, Toronto (March 28, 2025).

    \item \textbf{Section~\ref{sec:dichromatic}: No countable basis for Borel directed graphs of dichromatic number at least three}\\
    This new result completes the picture for all finite Borel chromatic and dichromatic thresholds.
    Presented at the Caltech Logic Seminar (April 29, 2026).

    \item \textbf{Section~\ref{sec:cantor_game}: Uncountable sets and an infinite linear order game}\\
    Adapted from \cite{MatosWiederholdSalvetti2025}, where we provide a solution to a problem originally posed by Brian and Clontz.
    Presented at the 2023 Summer Meeting of the Canadian Mathematical Society in Ottawa.

    \item \textbf{Chapter~\ref{ch:deep_computations}: Deep Computations}\\
    Primarily based on \cite{Duenezetal2026} and a multi-part series at the Fields Institute Set Theory Seminar (2024--2026).

    \item \textbf{Section~\ref{sec:border_tracing}: Border tracing in oriented adjacency graphs of polygonal tilings}\\
    Based on my contributions to \cite{WiederholdMatosWiederhold2026}.

    \item \textbf{Section~\ref{sec:recursive_constructions}: Recursive constructions}\\
    Extends joint work found in \cite{Eslavaetal2025} and \cite{Caroetal2023}.
    Presented at the 2022 Summer Meeting of the Canadian Mathematical Society in St John's.

    \item \textbf{Section~\ref{sec:amoeba_trees}: Amoeba trees}\\
    Based on \cite{GomezetalPreprint}.
\end{itemize}
\end{publications}
